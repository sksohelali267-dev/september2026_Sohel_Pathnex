## Jenkins Pipeline — Jenkinsfile

```groovy
pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Sohel"
    }

    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/sohel/sample-java-app.git'
            }
        }

        stage('Use Secret') {
            steps {
                withCredentials([
                    string(
                        credentialsId: 'SOHEL_API_KEY',
                        variable: 'API_KEY'
                    )
                ]) {
                    sh 'echo "Using secret for $INSTITUTE_NAME"'
                }
            }
        }
    }
}
```

## GitLab CI/CD — .gitlab-ci.yml

```yaml
stages:
  - build

variables:
  INSTITUTE_NAME: "Sohel"

build:
  stage: build
  image: alpine:latest
  script:
    - echo "Building application for $INSTITUTE_NAME"
    - |
      if [ -z "$API_KEY" ]; then
        echo "API_KEY is not configured"
        exit 1
      fi
      echo "API key is configured"
```
